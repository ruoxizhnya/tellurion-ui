import polars as pl
from quantlab.engine import AlphaSignal, Parameter

class CPOVolSurgeAlpha(AlphaSignal):
    lookback = Parameter(default=240, min=120, max=360)
    surge_threshold = Parameter(default=2.5)  # 波动率放大量倍数
    
    def compute_signals(self, df_bars: pl.DataFrame) -> pl.Series:
        rolling_std = df_bars["volume"].rolling_std(window_size=self.lookback)
        surge_event = df_bars["volume"] > (df_bars["volume"].rolling_mean(self.lookback) + self.surge_threshold * rolling_std)
        intraday_decay = pl.when(df_bars["time"].dt.hour() >= 14).then(0.85).otherwise(1.0)
        return (surge_event.cast(pl.Float32) * intraday_decay).shift(1)  # 严格T+1生效
